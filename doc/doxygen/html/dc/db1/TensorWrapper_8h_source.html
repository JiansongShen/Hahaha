220 T& at(const std::initializer_list<size_t>& indices) {
221 const auto& shapeDims = data_.getShape().getDims();
222 if (indices.size() != shapeDims.size()) {
223 throw std::out_of_range("Dimension mismatch: expected "
224 + std::to_string(shapeDims.size())
225 + " indices, got "
226 + std::to_string(indices.size()));
227 }
228
229 size_t linearIdx = 0;
230 const auto* idxIt = indices.begin();
231 const auto& strideDims = data_.getStride().getStrides();
232
233 auto dimsSize = shapeDims.size();
234 for (size_t i = 0; i < dimsSize; ++i) {
235 size_t dimIdx = *idxIt;
236 if (dimIdx >= shapeDims[i]) {
237 throw std::out_of_range("Index out of bounds at dimension "
238 + std::to_string(i));
239 }
240 linearIdx += dimIdx * strideDims[i];
241 std::advance(idxIt, 1);
242 }
243 return data_.getData()[linearIdx];
244 }

744 TensorWrapper sum(std::vector<size_t> axes,
745 const bool keepDims = false) const {
746 if (axes.empty()) {
747 return this->clone();
748 }
749
750 std::ranges::sort(axes);
751 axes.erase(std::ranges::unique(axes).begin(), axes.end());
752
753 // 1. get target shape
754 std::vector<bool> isReduced;
755 const std::vector<size_t>& srcShape = getShape();
756 std::vector<size_t> resShape;
757
758 isReduced.resize(srcShape.size(), false);
759 for (const unsigned long axe : axes) {
760 if (static_cast<long>(axe)
761 > static_cast<long>(isReduced.size()) - 1) {
762 throw std::invalid_argument("axis is too big!");
763 }
764 isReduced[axe] = true;
765 }
766
767 if (axes.size() == this->getShape().size()) {
768 TensorWrapper result;
769 result.data_.setShape(TensorShape({}));
770 result.data_.setStride(TensorStride(result.data_.getShape()));
771 result.data_.setData(std::make_shared<T[]>(1));
772 result.getRawData().get()[0] = this->sum();
773 return result;
774 }
775
776 if (this->getShape().size() == 0) {
777 return this->clone();
778 }
779
780 for (size_t i = 0; i < srcShape.size(); ++i) {
781 // if sum for this dim, then remove it or set 1 in resShape
782 if (isReduced[i]) {
783 if (keepDims) {
784 resShape.push_back(1);
785 }
786 } else {
787 // if not then just add dim
788 resShape.push_back(srcShape[i]);
789 }
790 }
791
792 // 2. calculate necessary datas
793 // need to calculate
794 // a. how many should dstIdx reduce when it needs to reduce
795 // when coord carries at a position(current value on position is a)
796 // if this pos is not reduced, then dstIdx should minus
797 // a * correspondStride
798 // if this pos is reduced, then do nothing,
799 // b. how many value should dstIdx add when it needs to add
800 // when coord increases at a position,
801 // if this pos is not reduced, then dstIdx should add a value
802 // equals to the stride
803 // if this pos is reduced, then dstIdx will add nothing
804 std::vector<size_t> resStride(srcShape.size(), 0);
805 TensorWrapper result((TensorShape(resShape)));
806 auto resultStride = result.getStride().getStrides();
807 size_t resultStrideIdx = 0;
808 for (size_t i = 0; i < srcShape.size(); ++i) {
809 if (isReduced[i]) {
810 if (keepDims) {
811 resultStrideIdx++;
812 }
813 } else {
814 resStride[i] = resultStride[resultStrideIdx++];
815 }
816 }
817
818 // 3. calculate data to result
819 const std::shared_ptr<T[]> srcPtr = getRawData();
820 std::shared_ptr<T[]> resPtr = result.getRawData();
821 std::vector<size_t> coord(srcShape.size(), 0);
822 size_t dstIdx = 0;
823
824 for (size_t srcIdx = 0; srcIdx < getTotalSize(); ++srcIdx) {
825 resPtr[dstIdx] += srcPtr[srcIdx];
826
827 for (long i = static_cast<long>(coord.size() - 1); i >= 0; --i) {
828 ++coord[i];
829 // need to carry, next value on position will add one
830 if (coord[i] == srcShape[i]) {
831 coord[i] = 0;
832 dstIdx -= resStride[i] * (srcShape[i] - 1);
833 continue;
834 }
835 dstIdx += resStride[i];
836 break;
837 }
838 }
839
840 return result;
841 }

878 TensorWrapper broadcastTo(const TensorShape& newShape) {
879
880 auto broadcasted =
881 TensorShape::broadcastShape(this->data_.getShape(), newShape);
882 if (!broadcasted.has_value() || TensorShape(*broadcasted) != newShape) {
883 throw std::invalid_argument("Cannot broadcast shape "
884 + this->data_.getShape().toString()
885 + " to " + newShape.toString());
886 }
887
888 TensorWrapper result;
889 result.data_ = this->data_.share();
890 if (newShape.getDims().size() < this->getShape().size()) {
891 throw std::invalid_argument(
892 "invalid argument of broadcastTo(), the target shape is "
893 + newShape.toString() + " but current shape is "
894 + this->data_.getShape().toString());
895 }
896
897 TensorStride newStride = this->data_.getStride();
898 auto shapeDiff = newShape.getDims().size() - this->getShape().size();
899 newStride.getStrides().insert(
900 newStride.getStrides().begin(), shapeDiff, 0);
901
902 long newShapeIdx = static_cast<long>(newShape.getDims().size() - 1);
903 long selfShapeIdx = static_cast<long>(this->getShape().size() - 1);
904
905 while (selfShapeIdx >= 0) {
906
907 if (newShape.getDims()[newShapeIdx] != 1
908 && getShape()[selfShapeIdx] == 1) {
909 newStride.getStrides()[newShapeIdx] = 0;
910 }
911
912 --newShapeIdx;
913 --selfShapeIdx;
914 }
915
916 result.data_.setShape(newShape);
917 result.data_.setStride(newStride);
918
919 return result;
920 }
