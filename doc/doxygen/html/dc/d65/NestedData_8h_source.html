66 NestedData(std::initializer_list<NestedData> data) {
67 if (data.size() == 0) {
68 return;
69 }
70
71 shape_.push_back(data.size());
72 const auto& firstNestedData = *data.begin();
73 // Check if all nested data have the same shape
74 for (const auto& val : data) {
75 if (val.shape_ != firstNestedData.shape_) {
76 throw std::invalid_argument("Nested initializer list has "
77 "inconsistent shapes.");
78 }
79 }
80 shape_.insert(shape_.end(),
81 firstNestedData.shape_.begin(),
82 firstNestedData.shape_.end());
83
84 size_t totalElements = 0;
85 HAHAHA_PRAGMA_UNROLL(5)
86 for (const auto& val : data) {
87 totalElements += val.flatData_.size();
88 }
89 flatData_.reserve(totalElements);
90
91 HAHAHA_PRAGMA_UNROLL(5)
92 for (const auto& val : data) {
93 flatData_.insert(
94 flatData_.end(), val.flatData_.begin(), val.flatData_.end());
95 }
96 }
97
102 const std::vector<T>& getFlatData() const {
103 return flatData_;
104 }

110 [[nodiscard]] const std::vector<size_t>& getShape() const {
111 return shape_;
112 }
