60 explicit Logger(LoggerConfig config)
61 : config_(std::move(config)), running_(true) {
62 if (config_.isWriteToFile()) {

71 ~Logger() {

173 void process() {
174 while (running_ || !queue_.empty()) {
175 // get log entry
176 LogMessageEntry entry;
177 {
178 std::unique_lock lock(mutex_);
179 condition_.wait(
180 lock, [this] { return !queue_.empty() || !running_; });
181
182 if (!running_ && queue_.empty()) {
183 break;
184 }
185
186 if (!queue_.empty()) {
187 entry = std::move(queue_.front());
188 queue_.pop();
189 } else {
190 continue;
191 }
192 }
193
194 std::string timestamp;
195 getTimeIfEnabled(timestamp);
196
197 if (config_.isWriteToFile()) {
198 stream_ << timestamp << "[" << toString(entry.getLevel()) << "]"
199 << entry.getMessage() << '\n';
200 stream_.flush(); // Ensure data is written
201 }
202 if (config_.isWriteToConsole()) {
203 std::cout << timestamp << "["
204 << toColoredString(entry.getLevel()) << "]"
205 << entry.getMessage() << '\n';
206 std::cout.flush(); // Ensure data is written
207 }
208 }
209 }
210
215 void getTimeIfEnabled(std::string& timestamp) {
216 if (config_.isEnableTime()) {
217 std::time_t now = std::time(nullptr);
218 std::tm localTime{};
219
220#ifdef _WIN32
221 localtime_s(&localTime, &now); // Windows
222#else
223 localtime_r(&now, &localTime); // POSIX
224#endif
225
226 std::string time =
227 std::format("{:04d}-{:02d}-{:02d} {:02d}:{:02d}:{:02d}",
228 localTime.tm_year + 1900,
229 localTime.tm_mon + 1,
230 localTime.tm_mday,
231 localTime.tm_hour,
232 localTime.tm_min,
233 localTime.tm_sec);
234 timestamp = std::format("[{}]", time);
235 }
236 }

247inline void Logger::log(const std::string& msg, LogLevel level) {
248 Logger& logger = instance();
249 {
250 std::scoped_lock const lock(logger.mutex_);
251 logger.queue_.emplace(level, msg);
252 }
253 logger.condition_.notify_one();
