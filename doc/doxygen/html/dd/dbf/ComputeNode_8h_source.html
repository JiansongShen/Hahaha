103 ComputeNode(std::shared_ptr<ComputeNode<T>> lhs,
104 std::shared_ptr<ComputeNode<T>> rhs,
105 std::shared_ptr<math::TensorWrapper<T>> res,
106 common::Operator operatorType,
107 std::function<void()> gradFun)
108 : data_(res), operatorType_(operatorType),
109 gradFun_(std::move(gradFun)) {
110 if (operatorType_ == common::Operator::None) {
111 throw std::invalid_argument("Operator cannot be None");
112 }
113 parents_.push_back(lhs);
114 parents_.push_back(rhs);
115 // Automatically determine if this node requires gradients based on
116 // parents
117 this->requiresGrad_ =
118 lhs->requiresGrad_ || rhs->requiresGrad_ || this->requiresGrad_;
119 }

136 void accumulateGrad(std::shared_ptr<math::TensorWrapper<T>> grad) {
137 if (this->grad_) {
138 // grad_total = grad_total + incoming_grad
139 this->grad_ = std::make_shared<math::TensorWrapper<T>>(
140 this->grad_->add(*grad));
141 } else {
142 // First gradient received, clone it to avoid side effects
143 this->grad_ = std::make_shared<math::TensorWrapper<T>>(*grad);
144 }
145 }

241 void backward() {
242 if (!requiresGrad_) {
243 return;
244 }
245
246 // 1. Get the topological sort of the graph ending at this node
247 TopoSort<T> sorter;
248 std::vector<std::shared_ptr<ComputeNode<T>>> topoList =
249 sorter.toTopoList(this->shared_from_this());
250
251 // 2. Initialize the gradient of the root node (e.g., Loss) to 1.0
252 if (!grad_) {
253 grad_ = std::make_shared<math::TensorWrapper<T>>(
254 math::TensorShape(data_->getShape()), 1.0f);
255 }
256
257 // 3. Iterate backwards through the topological list
258 // topoList is [Inputs..., Ops..., Output]
259 // reverse is [Output, Ops..., Inputs...]
260 for (auto it = topoList.rbegin(); it != topoList.rend(); ++it) {
261 // Execute the gradient function for this node to propagate to
262 // parents
263 if (auto& node = *it; node->gradFun_) {
264 node->gradFun_();
265 }
266 }
267 }

269 static std::shared_ptr<ComputeNode>
270 createUnary(std::shared_ptr<ComputeNode> parent,
271 std::shared_ptr<math::TensorWrapper<T>> res,
272 common::Operator operatorType,
273 std::function<void()> gradFun = nullptr) {
274 std::shared_ptr<ComputeNode> node = std::make_shared<ComputeNode>(
275 res, operatorType, std::move(gradFun));
276 node->addParent(parent);
277 node->setRequiresGrad(parent->getRequiresGrad());
278 return node;
279 }

281 [[nodiscard]] common::Operator getOperatorType() const {
