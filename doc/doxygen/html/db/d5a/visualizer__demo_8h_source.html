39inline void run_visualizer_demo() {
40 using namespace hahaha::display;
41
42 // 1. Create and initialize the window (handles GLFW/OpenGL/ImGui initialization)
43 GlfwWindow window;
44 try {
45 window.init("Hahaha ML Training Demo", 1280, 720);
46 } catch (const std::exception& e) {
47 std::cerr << "Failed to initialize window: " << e.what() << std::endl;
48 return;
49 }
50
51 // 2. Create the visualizer
52 auto visualizer = createMLVisualizer();
53
54 // 3. Define our "Mock" model structure
55 visualizer->addLayer({"Input (Images)", 784, 256});
56 visualizer->addLayer({"ReLU", 256, 256});
57 visualizer->addLayer({"Hidden 1", 256, 128});
58 visualizer->addLayer({"ReLU", 128, 128});
59 visualizer->addLayer({"Hidden 2", 128, 64});
60 visualizer->addLayer({"Sigmoid", 64, 64});
61 visualizer->addLayer({"Output (Classes)", 64, 10});
62
63 visualizer->setStatus("Ready to train");
64
65 int epoch = 0;
66 bool training = true;
67 float currentLoss = 2.5f;
68 float currentAcc = 0.1f;
69
70 std::cout << "Starting Visualizer loop with a real GLFW window." << std::endl;
71
72 // 4. Main Loop
73 while (window.render()) {
74
75 // Handle User Interaction
76 ControlAction action = visualizer->getControlAction();
77 if (action == ControlAction::Start) {
78 training = true;
79 visualizer->setStatus("Training in progress...");
80 } else if (action == ControlAction::Pause) {
81 training = false;
82 visualizer->setStatus("Training paused");
83 } else if (action == ControlAction::Stop) {
84 std::cout << "User requested stop via UI." << std::endl;
85 break;
86 } else if (action == ControlAction::Reset) {
87 epoch = 0;
88 currentLoss = 2.5f;
89 currentAcc = 0.1f;
90 visualizer->setStatus("Reset completed");
91 }
92
93 // Simulate Training Step
94 if (training) {
95 currentLoss *= 0.995f;
96 currentAcc = 1.0f - (currentLoss / 2.5f);
97
98 if (epoch < 1000 && (frameCount % 10 == 0)) {
99 visualizer->recordMetrics(epoch++, currentLoss, currentAcc);
100 if (epoch % 10 == 0) {
101 std::cout << "[Demo] Epoch: " << epoch << ", Loss: " << currentLoss << ", Acc: " << currentAcc << std::endl;
102 }
103 }
104 }
105 frameCount++;
106
107 // Render the UI widgets
108 visualizer->renderFrame();
109
110 // Finalize rendering
111 ImGui::Render();
112 int display_w, display_h;
113 glfwGetFramebufferSize(window.getHandle(), &display_w, &display_h);
114 glViewport(0, 0, display_w, display_h);
115 glClearColor(0.1f, 0.1f, 0.12f, 1.00f);
116 glClear(GL_COLOR_BUFFER_BIT);
117 ImGui_ImplOpenGL3_RenderDrawData(ImGui::GetDrawData());
118
119 glfwSwapBuffers(window.getHandle());
120
121 // Small sleep to control frame rate if vsync is off or for simulation stability
122 std::this_thread::sleep_for(std::chrono::milliseconds(1));
123 }
124}
125
126} // namespace hahaha::example
127
128#endif // HAHAHA_EXAMPLE_ML_VISUALIZER_DEMO_H
