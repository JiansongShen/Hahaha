79template <typename T>
80std::shared_ptr<ComputeNode<T>>
81broadcast(const std::shared_ptr<ComputeNode<T>>& sourceNode,
82 const std::vector<size_t>& targetTensorShape) {
83
84 checkTensorCanBroadcastTo(sourceNode, targetTensorShape);
85
86 auto resNode = ComputeNode<T>::createUnary(
87 sourceNode,
88 std::make_shared<math::TensorWrapper<T>>(
89 sourceNode->getData()->broadcastTo(TensorShape(targetTensorShape))),
90 common::Operator::Broadcast,
91 nullptr);
92
93 std::weak_ptr<ComputeNode<T>> weakRes = resNode;
94 std::weak_ptr<ComputeNode<T>> weakSrc = sourceNode;
95
96 resNode->setGradFun([weakSrc, weakRes] {
97 auto res = weakRes.lock();
98 auto src = weakSrc.lock();
99 if (res && src) {
100 std::shared_ptr<math::TensorWrapper<T>> resData = res->getData();
101 std::shared_ptr<math::TensorWrapper<T>> srcData = src->getData();
102 // detect shape different
103 std::vector<size_t> sumAxes;
104 const std::vector<size_t> srcShape = srcData->getShape();
105 const std::vector<size_t> resShape = resData->getShape();
106 bool keepDims = resShape.size() == srcShape.size();
107
108 // srcIdx must smaller than resIdx
109 long srcIdx = static_cast<long>(srcShape.size() - 1);
110 long resIdx = static_cast<long>(resShape.size() - 1);
111
112 while (srcIdx >= 0 && resIdx >= 0) {
113 if (srcShape[srcIdx] != resShape[resIdx]) {
114 sumAxes.push_back(resIdx);
115 }
116 --srcIdx;
117 --resIdx;
118 }
119
120 while (resIdx >= 0) {
121 sumAxes.push_back(resIdx);
122 --resIdx;
123 }
124
125 auto grad = res->getGrad();
126 if (!grad) {
127 return;
128 }
129 auto srcGrad = grad->sum(sumAxes, keepDims);
130 src->accumulateGrad(
131 std::make_shared<math::TensorWrapper<T>>(std::move(srcGrad)));
132 }
133 });
134
135 return resNode;
136}

138template <typename T>
139void checkTensorCanBroadcastTo(
140 const std::shared_ptr<ComputeNode<T>>& sourceNode,
141 const std::vector<size_t>& targetTensorShape) {
142
143 const auto srcShape = TensorShape(sourceNode->getData()->getShape());
144 const TensorShape targetShape(targetTensorShape);
145 const auto broadcastShape =
146 TensorShape::broadcastShape(srcShape, targetShape);
147 if (!broadcastShape.has_value()
148 || broadcastShape.value() != targetTensorShape) {
149 std::string broadcastShapeStr = broadcastShape.has_value()
150 ? TensorShape(broadcastShape.value()).toString()
151 : "None";
152 throw std::runtime_error(
153 "broadcast shape mismatch: source tensor shape is "
154 + srcShape.toString() + ", \n\tbut target tensor shape is "
155 + targetShape.toString()
156 + "\tdo you want to broadcast to :" + broadcastShapeStr);
157 }
158}

176template <typename T>
177std::pair<std::shared_ptr<ComputeNode<T>>, std::shared_ptr<ComputeNode<T>>>
178broadcastNodes(const std::shared_ptr<ComputeNode<T>>& lhs,
179 const std::shared_ptr<ComputeNode<T>>& rhs) {
180 const auto lhsShape = TensorShape(lhs->getData()->getShape());
181 const auto rhsShape = TensorShape(rhs->getData()->getShape());
182
183 if (lhsShape == rhsShape) {
184 return {lhs, rhs};
185 }
186
187 const auto commonShape = TensorShape::broadcastShape(lhsShape, rhsShape);
188 if (!commonShape.has_value()) {
189 throw std::invalid_argument("Shapes are not broadcast-compatible: "
190 + lhsShape.toString() + " and "
191 + rhsShape.toString());
192 }
193
194 auto targetShape = commonShape.value();
195 auto newLhs = lhs;
196 auto newRhs = rhs;
197
198 if (lhsShape.getDims() != targetShape) {
199 newLhs = broadcast(lhs, targetShape);
200 }
201 if (rhsShape.getDims() != targetShape) {
202 newRhs = broadcast(rhs, targetShape);
203 }
204
205 return {newLhs, newRhs};
206}
