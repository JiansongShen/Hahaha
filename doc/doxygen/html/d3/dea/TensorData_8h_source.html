60 TensorData(const TensorShape& shape,
61 T initValue,
62 const backend::Device device = backend::Device())
63 : shape_(shape), stride_(shape), device_(device) {
64 size_t size = shape_.getTotalSize();
65 if (device_.type == backend::DeviceType::CPU
66 || device_.type == backend::DeviceType::SIMD) {
67 data_ = std::make_shared<T[]>(size);
68 std::fill(data_.get(), data_.get() + size, initValue);
69 } else {
70 // TODO: Handle GPU allocation using compute::gpu::GpuMemory
71 throw std::runtime_error(
72 "GPU allocation not yet implemented in TensorData");
73 }
74 }

98 TensorData(const TensorData& other)

100 size_t size = shape_.getTotalSize();
101 if (device_.type == backend::DeviceType::CPU

103 data_ = std::make_shared<T[]>(size);
104 std::copy(other.data_.get(), other.data_.get() + size, data_.get());
105 } else {
106 // TODO: Handle GPU deep copy

126 explicit TensorData(const std::vector<T>& initVec)

181 explicit TensorData(NestedData<T>&& data) : shape_(data.getShape()) {
182 size_t size = data.getFlatData().size();
183 if (size > 0) {
184 data_ = std::make_shared<T[]>(size);
185 std::copy(data.getFlatData().begin(),
186 data.getFlatData().end(),
187 data_.get());

189 data_ = nullptr; // Explicitly null for truly empty tensors
