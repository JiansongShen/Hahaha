39template <typename T>
40std::shared_ptr<ComputeNode<T>>
41reshape(const std::shared_ptr<ComputeNode<T>>& parent,
42 const std::vector<size_t>& newShape) {
43 auto resData = std::make_shared<math::TensorWrapper<T>>(
44 parent->getData()->reshape(newShape));
45
46 std::shared_ptr<ComputeNode<T>> resNode =
47 ComputeNode<T>::createUnary(parent, resData, common::Operator::Reshape);
48
49 std::weak_ptr<ComputeNode<T>> weakRes = resNode;
50 std::weak_ptr<ComputeNode<T>> weakParent = parent;
51 auto originalShape = parent->getData()->getShape();
52
53 resNode->setGradFun([weakParent, weakRes, originalShape]() {
54 auto res = weakRes.lock();
55 auto p = weakParent.lock();
56 if (res && p) {
57 auto gradPtr = res->getGrad();
58 if (p->getRequiresGrad()) {
59 auto reshapedGrad = std::make_shared<math::TensorWrapper<T>>(
60 gradPtr->reshape(originalShape));
61 p->accumulateGrad(reshapedGrad);
62 // p->backward();
63 }
64 }
65 });
66 return resNode;
67}

111template <typename T>
112std::shared_ptr<ComputeNode<T>>
113neg(const std::shared_ptr<ComputeNode<T>>& parent) {
114 auto resData =
115 std::make_shared<math::TensorWrapper<T>>(-*parent->getData());
116 auto resNode =
117 ComputeNode<T>::createUnary(parent, resData, common::Operator::Neg);
118
119 std::weak_ptr<ComputeNode<T>> weakRes = resNode;
120 std::weak_ptr<ComputeNode<T>> weakParent = parent;
121
122 resNode->setGradFun([weakParent, weakRes]() {
123 auto res = weakRes.lock();
124 auto parent = weakParent.lock();
125 auto gradPtr = res->getGrad();
126 if (parent->getRequiresGrad()) {
127 auto negatedGrad =
128 std::make_shared<math::TensorWrapper<T>>(-*gradPtr);
129 parent->accumulateGrad(negatedGrad);
130 }
131 });
132 return resNode;
133}
