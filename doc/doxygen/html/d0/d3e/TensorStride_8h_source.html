53 template <typename T> explicit TensorStride(const std::vector<T>& dims) {
54 strides_.resize(dims.size());
55 if (dims.empty()) {
56 return;
57 }
58 strides_[dims.size() - 1] = 1;
59 HAHAHA_PRAGMA_UNROLL(5)
60 for (size_t i = dims.size() - 1; i > 0; --i) {
61 strides_[i - 1] = strides_[i] * static_cast<size_t>(dims[i]);
62 }
63 }
