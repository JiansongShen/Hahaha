159 [[nodiscard]] static std::optional<std::vector<size_t>>
160 broadcastShape(const TensorShape& shape1, const TensorShape& shape2) {
161 std::optional result = std::vector<size_t>{};
162
163 long shapeIdx1 = static_cast<long>(shape1.getDims().size() - 1);
164 long shapeIdx2 = static_cast<long>(shape2.getDims().size() - 1);
165
166 const auto& dims1 = shape1.getDims();
167 const auto& dims2 = shape2.getDims();
168
169 while (shapeIdx1 >= 0 || shapeIdx2 >= 0) {
170
171 auto d1 = shapeIdx1 < 0 ? 1 : dims1[shapeIdx1];
172 auto d2 = shapeIdx2 < 0 ? 1 : dims2[shapeIdx2];
173
174 if (d1 == d2 || d2 == 1) {
175 result->push_back(d1);
176 } else if (d1 == 1) {
177 result->push_back(d2);
178 } else if (d1 != d2) {
179 return std::nullopt;
180 }
181 shapeIdx1--;
182 shapeIdx2--;
183 }
184
185 std::ranges::reverse(*result);
186 return result;
187 }
