33inline void nn_train_example() {
34
35 std::mt19937 engine;
36
37 std::random_device randomDevice;
38 std::seed_seq seed{
39 randomDevice(),
40 static_cast<unsigned int>(
41 std::chrono::steady_clock::now().time_since_epoch().count())};
42 engine.seed(seed);
43
44 std::uniform_real_distribution<float> dist(0, 100);
45
46 constexpr int DataSize = 100;
47 constexpr int TrainLoop = 10;
48
49 std::vector<f32> x;
50 std::vector<f32> y;
51 x.resize(DataSize);
52 y.resize(DataSize);
53
54 // y = 2 * x
55 for (int i = 0; i < DataSize; ++i) {
56 auto val = dist(engine);
57 x[i] = val;
58 y[i] = val * 2;
59 }
60
61 Tensor<f32> const xTensor1 = Tensor<f32>::buildFromVector(x);
62 Tensor<f32> const yTensor1 = Tensor<f32>::buildFromVector(y);
63 Tensor<f32> const xTensorM = xTensor1.reshape({100, 1});
64 Tensor<f32> const yTensorM = yTensor1.reshape({100, 1});
65 Tensor<f32> w(1);
66 w.setRequiresGrad(true);
67 // y = w * x
68 Tensor<f32> loss(1);
69 loss.setRequiresGrad(true);
70
71 auto optimizer = hahaha::ml::SGDOptimizer<f32>({w}, 0.000001);
72
73 for (int i = 0; i < TrainLoop; ++i) {
74 optimizer.zeroGrad();
75 auto tmp = yTensorM - w * xTensorM;
76
77 loss = (yTensorM - w * xTensorM)
78 .transpose()
79 .matmul(yTensorM - w * xTensorM);
80
81 loss.backward();
82 optimizer.step();
83
84 std::cout << "epoch: " << 1 + i << std::endl;
85 std::cout << " w: " << w.at({}) << std::endl;
86 std::cout << " loss: " << loss.data()->at({0,0}) << std::endl;
87 }
88}
